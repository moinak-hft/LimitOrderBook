cmake_minimum_required(VERSION 3.20)
project(limit_order_book VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LOB_NATIVE "Optimize for the compiling CPU (less portable binary)" OFF)
option(LOB_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(LOB_BUILD_BASELINE "Build the preserved original for local comparisons" ON)
find_package(Threads REQUIRED)
add_library(lob_options INTERFACE)
target_include_directories(lob_options INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(lob_options INTERFACE /W4 /permissive- /EHsc /utf-8)
else()
  target_compile_options(lob_options INTERFACE -Wall -Wextra -Wpedantic)
  if(LOB_NATIVE)
    target_compile_options(lob_options INTERFACE -march=native)
  endif()
  if(LOB_SANITIZERS)
    target_compile_options(lob_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(lob_options INTERFACE -fsanitize=address,undefined)
  endif()
endif()
add_executable(limit_order_book main.cpp Benchmark.cpp ReplayLab.cpp)
target_link_libraries(limit_order_book PRIVATE lob_options Threads::Threads)
# Embed the offline report so the executable works from any working directory.
# Template edits automatically regenerate the header on the next build.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/replay.html")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/replay.html" LOB_REPORT_HTML)
# Keep individual literals small for MSVC; preserve each source newline exactly.
string(REPLACE "\n" "\n)LOB_REPORT\"\nR\"LOB_REPORT(" LOB_REPORT_HTML_CHUNKS "${LOB_REPORT_HTML}")
configure_file(assets/ReplayTemplate.h.in generated/ReplayTemplate.h @ONLY)
target_include_directories(limit_order_book PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(LOB_BUILD_BASELINE)
  add_executable(original_benchmark original/main.cpp)
  target_link_libraries(original_benchmark PRIVATE lob_options)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(engine_tests tests/engine_tests.cpp)
  target_link_libraries(engine_tests PRIVATE lob_options)
  add_test(NAME engine_correctness COMMAND engine_tests)
  add_test(NAME simulation_smoke COMMAND limit_order_book simulate --steps 48 --quantity 1200 --seed 42 --output ${CMAKE_CURRENT_BINARY_DIR}/test-reports/replay.html)
  add_test(NAME cli_help COMMAND limit_order_book --help)
  add_test(NAME cli_rejects_invalid COMMAND limit_order_book simulate --steps 0)
  set_tests_properties(cli_rejects_invalid PROPERTIES WILL_FAIL TRUE)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME report_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/verify_reports.py $<TARGET_FILE:limit_order_book>)
  endif()
endif()
